# Lower bounds are the versions this project is developed and tested against;
# upper bounds keep a major release from breaking the app unattended.
#
# pandas >= 2.2 is a hard floor: the code uses the lowercase frequency aliases
# ("h", "min") that were introduced there.
#
# urllib3 is deliberately NOT pinned below 2.0. pytrends only breaks on
# urllib3 2.x if you pass `retries`/`backoff_factor` to TrendReq (it forwards
# them to urllib3.Retry using the removed `method_whitelist` kwarg).
# TrendsAnalyzer avoids that and runs its own retry loop, so urllib3 2.x is
# fine — see the note in analyzer/trends.py.

streamlit>=1.28.0,<2.0
plotly>=5.18.0,<7.0
pandas>=2.2.0,<3.0
numpy>=1.24.0,<3.0
scikit-learn>=1.3.0,<2.0
statsmodels>=0.14.0,<1.0
